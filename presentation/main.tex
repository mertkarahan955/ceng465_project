\section{Motivation \& Scope}

\begin{frame}{What Is This Project About?}
  \begin{columns}
    \column{0.55\textwidth}
      \textbf{Goal:} Build and observe a \alert{single-leader replication} system across two physical machines.

      \vspace{0.4cm}
      \textbf{Required behavior:}
      \begin{itemize}
        \item One node acts as \textcolor{primary}{Leader (Primary)} --- receives all writes
        \item One node acts as \textcolor{secondary}{Follower (Secondary)} --- replicates from leader
        \item Measure \alert{when} and \alert{how quickly} writes propagate
        \item Log every operation with ordering metadata
      \end{itemize}

      \vspace{0.4cm}
      \textbf{Deployment:} Two separate macOS machines on the same LAN --- satisfies the project's \emph{truly distributed} requirement.

    \column{0.42\textwidth}
      \begin{block}{Project Scope (Progress)}
        \begin{enumerate}
          \item \textbf{Environment Setup}\\
                Replica set on two machines
          \item \textbf{Schema Design}\\
                Replication-aware data model
          \item \textbf{Replication Logging}\\
                Operation logs with timestamps
          \item \textbf{Insert / Update / Delete}\\
                Visibility and delay tracking
        \end{enumerate}
      \end{block}
  \end{columns}
\end{frame}

% ─── Section 2 ───────────────────────────────────────────────────────────────
\section{Why MongoDB?}

\begin{frame}{Database Choice: MongoDB Replica Set}
  \begin{columns}
    \column{0.5\textwidth}
      \textbf{Key requirements from the project:}
      \begin{itemize}
        \item Single-leader replication
        \item Observable update propagation
        \item Support for insert, update, delete
        \item Replication lag must be measurable
      \end{itemize}

      \vspace{0.3cm}
      \textbf{MongoDB Replica Set provides all of this natively:}
      \begin{itemize}
        \item \textcolor{primary}{Primary} node accepts writes
        \item \textcolor{secondary}{Secondary} applies operations from the \textbf{oplog}
        \item Replication is \alert{asynchronous} --- lag is observable
        \item \texttt{rs.status()} exposes health and lag in real time
      \end{itemize}

    \column{0.47\textwidth}
      \begin{alertblock}{Raft-Inspired Metadata}
        Rather than implementing full Raft, we embed Raft-style tracking fields directly in our documents:
        \begin{itemize}
          \item \texttt{leader\_term}
          \item \texttt{log\_index}
          \item \texttt{operation\_id}
          \item \texttt{version}
        \end{itemize}
        This allows analysis of \textbf{update ordering} and \textbf{visibility} without replacing MongoDB's built-in replication.
      \end{alertblock}
  \end{columns}
\end{frame}

\section{Setup Challenges}

\begin{frame}{Installation Challenges}
  \begin{block}{Challenge 1: Duplicate \texttt{bindIp} in \texttt{mongod.conf}}
    The Homebrew default config and our LAN IP entry were both present under \texttt{net:}, causing \texttt{mongod} to fail on startup silently.\\
    \textbf{Fix:} Removed duplicate key, kept only \texttt{bindIp: 127.0.0.1,192.168.88.146}
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Challenge 2: Non-Breaking Spaces in Config File}
    Copying YAML from a chat message introduced Unicode non-breaking spaces (\texttt{0xC2A0}) instead of ASCII spaces. \texttt{mongod} reported \texttt{"Unrecognized option: systemLog"}.\\
    \textbf{Fix:} Rewritten with \texttt{cat > mongod.conf << 'EOF'} to guarantee ASCII encoding.
  \end{block}

  \vspace{0.2cm}
  \begin{block}{Challenge 3: \texttt{sudo brew services} Ownership Issue}
    Running \texttt{sudo brew services} changed file ownership to root, preventing the service from starting as a user process.\\
    \textbf{Fix:} \texttt{sudo chown -R \$(whoami)} on affected paths, then restarted without sudo.
  \end{block}
\end{frame}

\begin{frame}{Installation Challenges (cont.)}
  \begin{block}{Challenge 4: Missing Log and Data Directories}
    \texttt{mongod} failed silently because \texttt{/opt/homebrew/var/log/mongodb/} did not exist yet.\\
    \textbf{Fix:} \texttt{mkdir -p} for both log and data directories before starting the service.
  \end{block}

  \vspace{0.3cm}
  \begin{block}{Challenge 5: macOS AirPlay Receiver on Port 5000}
    Flask defaulted to port 5000, which is reserved by macOS AirPlay Receiver (ControlCenter process).\\
    \textbf{Fix:} Moved Flask dashboard to port \texttt{5001}.
  \end{block}

  \vspace{0.3cm}
  \begin{block}{Challenge 6: Secondary Not Reachable from Primary}
    Secondary Mac's \texttt{mongod} was binding only to \texttt{127.0.0.1} and not the LAN IP, making port 27017 unreachable from the primary.\\
    \textbf{Fix:} Corrected \texttt{bindIp} to include \texttt{192.168.88.105}, recreated missing directories, restarted service.
  \end{block}
\end{frame}

\section{Schema Design}

\begin{frame}[fragile]{Schema: \texttt{items} Collection}
  Stores application data with replication-aware tracking fields.

  \vspace{0.2cm}
  \begin{lstlisting}[language=Python]
{
  "_id":              ObjectId,
  "key":              string,          # human-readable identifier
  "value":            object,          # application payload
  "version":          int,             # increments on every write
  "leader_term":      int,             # Raft-inspired term number
  "last_log_index":   int,             # global operation order
  "last_operation_id": UUID string,    # unique per write
  "last_updated":     ISODate,         # timestamp of last write on leader
  "deleted":          bool,            # soft-delete flag
  "created_by":       string           # client identifier
}
  \end{lstlisting}

  \vspace{0.1cm}
  \begin{columns}
    \column{0.5\textwidth}
      \textbf{Why \texttt{version}?}\\
      Allows the control node to poll the secondary and detect \emph{exactly} when a specific write becomes visible --- without relying on timestamps alone.

    \column{0.5\textwidth}
      \textbf{Why soft delete?}\\
      Preserves the document on secondary for replication analysis. Hard deletes would remove the evidence of propagation timing.
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Schema: \texttt{operation\_logs} Collection}
  Every write produces one log entry. This collection is the primary source of evidence for replication behavior.

  \vspace{0.2cm}
  \begin{lstlisting}[language=Python]
{
  "operation_id":          UUID string,
  "leader_term":           int,
  "log_index":             int,                  # global ordering
  "operation_type":        "insert" | "update" | "delete",
  "target_id":             ObjectId,
  "leader_write_time":     ISODate,              # write confirmed on primary
  "follower_visible_time": ISODate | null,       # secondary returned correct version
  "replication_delay_ms":  float | null,         # follower_visible - leader_write
  "version_before":        int | null,
  "version_after":         int,
  "status":                "visible_on_follower" | "timeout"
}
  \end{lstlisting}

  \vspace{0.1cm}
  \begin{alertblock}{Key Insight}
    \texttt{replication\_delay\_ms = follower\_visible\_time - leader\_write\_time}\\
    This is measured by polling the secondary every 10ms until \texttt{version == version\_after}.
  \end{alertblock}
\end{frame}

\section{Implementation}

\begin{frame}{Implementation Overview}
  \begin{columns}
    \column{0.5\textwidth}
      \begin{block}{Control Node (Python)}
        \begin{itemize}
          \item \textbf{config.py} --- hosts, ports, timeouts
          \item \textbf{db.py} --- primary \& secondary connections
          \item \textbf{operations.py} --- insert/update/delete with logging
          \item \textbf{app.py} --- Flask web dashboard (port 5001)
          \item \textbf{benchmark.py} --- 1000-op stress test
          \item \textbf{test\_replication.py} --- 13 assertions
          \item \textbf{trace.py} --- live change stream watcher
        \end{itemize}
      \end{block}

    \column{0.47\textwidth}
      \begin{block}{Write Flow (every operation)}
        \begin{enumerate}
          \item Write to primary with \texttt{w="majority"}
          \item Record \texttt{leader\_write\_time}
          \item Poll secondary every 10ms
          \item When \texttt{version} matches: record \texttt{follower\_visible\_time}
          \item Compute \texttt{replication\_delay\_ms}
          \item Insert into \texttt{operation\_logs}
        \end{enumerate}
      \end{block}

      \vspace{0.2cm}
      \textbf{Timeout:} 5000ms --- if secondary does not show the update within 5 seconds, the log entry status is set to \texttt{"timeout"}.
  \end{columns}
\end{frame}

\begin{frame}{Replica Set Verification}
  After both machines were running, the replica set was initialized from the primary:

  \vspace{0.3cm}
  \begin{columns}
    \column{0.5\textwidth}
      \textbf{\texttt{rs.initiate()} command:}
      \begin{itemize}
        \item \texttt{\_id: "rs0"}
        \item Member 0: \texttt{mongo-primary.lan:27017} (priority 2)
        \item Member 1: \texttt{mongo-secondary.lan:27017} (priority 1)
      \end{itemize}

    \column{0.47\textwidth}
      \begin{alertblock}{Verified result}
        \texttt{mongo-primary.lan:27017} \textcolor{primary}{PRIMARY}\\
        \texttt{mongo-secondary.lan:27017} \textcolor{secondary}{SECONDARY}
      \end{alertblock}
  \end{columns}

  \vspace{0.4cm}
  \begin{block}{Write concern: \texttt{w="majority"}}
    Every write operation uses \texttt{WriteConcern(w="majority")}, which means MongoDB waits for the primary \emph{and} at least one secondary to acknowledge the write before confirming to the client.
  \end{block}
\end{frame}
