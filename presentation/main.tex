% ── Slide 2: Project Overview ─────────────────────────────────────────────────
\begin{frame}{Project Overview}
  \begin{columns}
    \column{0.54\textwidth}
      \textbf{Goal:} demonstrate single-leader replication on a truly distributed setup --- two physical macOS machines, no containers.

      \vspace{0.35cm}
      \textbf{What we built}
      \begin{itemize}
        \item MongoDB replica set on two machines (LAN)
        \item Replication-aware data schema
        \item Operation logging (insert / update / delete)
        \item Write concern toggle: \texttt{w=majority} vs \texttt{w=1}
        \item Dashboard: health, delay, and sync state
      \end{itemize}

    \column{0.42\textwidth}
      \begin{block}{Central Question}
        What happens when the follower is unavailable?
        \begin{itemize}
          \item Should a write \textbf{block}?
          \item Should it be \textbf{queued} on the leader?
          \item How do we \textbf{prove} catch-up?
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

% ── Slide 4: Replica Set Configuration ───────────────────────────────────────
\begin{frame}[fragile]{Replica Set Configuration}
  \begin{columns}
    \column{0.5\textwidth}
      \textbf{Initial issue}
      \begin{itemize}
        \item Two voting members: primary + secondary
        \item Disconnecting secondary removed majority
        \item MongoDB reported \texttt{ReplicaSetNoPrimary}
        \item Even \texttt{w=1} could not write --- no writable primary
      \end{itemize}

    \column{0.46\textwidth}
      \begin{alertblock}{Solution: non-voting read replica}
        \begin{itemize}
          \item Primary: \texttt{votes=1, priority=2}
          \item Secondary: \texttt{votes=0, priority=0}
        \end{itemize}
        Primary stays writable during follower outages.
      \end{alertblock}
  \end{columns}

  \vspace{0.25cm}
  \begin{lstlisting}[language={}]
cfg = rs.conf()
cfg.members[1].priority = 0
cfg.members[1].votes = 0
rs.reconfig(cfg)
  \end{lstlisting}
\end{frame}

% ── Slide 6: Write Concern Modes ──────────────────────────────────────────────
\begin{frame}{Write Concern Modes}
  \begin{table}
    \small
    \begin{tabular}{p{0.18\textwidth}p{0.35\textwidth}p{0.35\textwidth}}
      \toprule
      Mode & Secondary \textbf{Up} & Secondary \textbf{Down} \\
      \midrule
      \texttt{w=majority} &
      Write succeeds; replication delay is measured and logged &
      \textbf{Write rejected} before primary mutation --- no data change \\
      \addlinespace
      \texttt{w=1} &
      Write succeeds immediately; log is reconciled in background &
      Write succeeds on primary; log stays \texttt{pending\_follower} until catch-up \\
      \bottomrule
    \end{tabular}
  \end{table}

  \vspace{0.2cm}
  \begin{columns}
    \column{0.48\textwidth}
      \begin{alertblock}{\texttt{w=majority} --- Consistency-first}
        Reject the write rather than risk divergence.
      \end{alertblock}
    \column{0.48\textwidth}
      \begin{block}{\texttt{w=1} --- Availability-first}
        Accept the write; guarantee eventual delivery via oplog.
      \end{block}
  \end{columns}
\end{frame}

% ── Slide 7: Async Recovery Flow ─────────────────────────────────────────────
\begin{frame}{Async Recovery Flow (\texttt{w=1})}
  \begin{columns}
    \column{0.58\textwidth}
      \begin{enumerate}
        \item User switches to \texttt{w=1} on the dashboard.
        \item Secondary network is disconnected.
        \item Write goes to primary; log records \texttt{pending\_follower}.
        \item \textit{Already-synced rows remain marked synced} (proven via prior logs).
        \item Secondary reconnects.
        \item MongoDB replays the oplog; secondary catches up.
        \item Control node reconciler detects version / log-index match.
        \item Log becomes \texttt{visible\_on\_follower}; sync checkmark appears.
      \end{enumerate}

    \column{0.38\textwidth}
      \begin{block}{Key design decisions}
        \begin{itemize}
          \item Secondary health uses \texttt{directConnection=true}
          \item Reconciler runs every second in background thread
          \item Fallbacks: version $\geq$ expected \textbf{or} log-index $\geq$ expected
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

% ── Slide 10: Results and Conclusion ──────────────────────────────────────────
\begin{frame}{Results and Conclusion}
  \begin{columns}
    \column{0.5\textwidth}
      \textbf{All manual tests passed}
      \begin{itemize}
        \item \texttt{w=majority} + secondary up: write visible on follower
        \item \texttt{w=majority} + secondary down: write rejected, no primary mutation
        \item \texttt{w=1} + secondary down: primary write succeeds (\texttt{pending\_follower})
        \item Reconnect: pending logs reconciled, sync restored
        \item Old synced rows stay synced during outage
      \end{itemize}

    \column{0.46\textwidth}
      \begin{block}{What we proved}
        \begin{itemize}
          \item Replication is \textbf{asynchronous by default}
          \item Write concern controls the \textbf{durability--availability tradeoff}
          \item Catch-up is driven by the \textbf{MongoDB oplog}, not the application
          \item \textbf{Observability} (version, log-index, delay) is essential for debugging distributed state
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}
